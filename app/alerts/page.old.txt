"use client";

import { useEffect, useRef, useState } from "react";

type Sig = {
  symbol: string;
  interval: string;
  price: number;
  ema9: number;
  ema21: number;
  rsi: number;
  signal: "UP" | "DOWN" | "NONE";
  candleTime: number;
};
type Entry = { time: string; text: string };

export default function Alerts() {
  const [symbol, setSymbol] = useState("BTCUSDT");
  const [interval, setIntervalValue] = useState("1m");
  const [running, setRunning] = useState(false);
  const [last, setLast] = useState<Sig | null>(null);
  const [error, setError] = useState("");
  const [log, setLog] = useState<Entry[]>([]);
  const alerted = useRef(0);

  function beep() {
    try {
      const ctx = new AudioContext();
      const o = ctx.createOscillator();
      o.frequency.value = 880;
      o.connect(ctx.destination);
      o.start();
      setTimeout(() => {
        o.stop();
        ctx.close();
      }, 500);
    } catch {}
  }

  function announce(s: Sig) {
    const text = `Alert: ${s.signal} signal on ${s.symbol}, ${s.interval} chart. Price ${s.price}.`;
    setLog((l) => [{ time: new Date().toLocaleTimeString(), text }, ...l].slice(0, 20));
    beep();
    if ("speechSynthesis" in window) {
      speechSynthesis.cancel();
      speechSynthesis.speak(new SpeechSynthesisUtterance(text));
    }
    if ("Notification" in window && Notification.permission === "granted") {
      new Notification("Jarvis alert", { body: text });
    }
  }

  useEffect(() => {
    if (!running) return;
    let stop = false;
    async function check() {
      try {
        const r = await fetch(`/api/signal?symbol=${symbol}&interval=${interval}`);
        const d = await r.json();
        if (stop) return;
        if (d.error) {
          setError(d.error);
          return;
        }
        setError("");
        setLast(d);
        if (d.signal !== "NONE" && d.candleTime !== alerted.current) {
          alerted.current = d.candleTime;
          announce(d);
        }
      } catch {
        setError("Could not reach the server.");
      }
    }
    check();
    const t = setInterval(check, 20000);
    return () => {
      stop = true;
      clearInterval(t);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [running, symbol, interval]);

  function toggle() {
    if (!running) {
      alerted.current = 0;
      if ("Notification" in window) Notification.requestPermission();
      beep();
    }
    setRunning(!running);
  }

  return (
    <div className="min-h-screen bg-[#050b18] p-6 text-cyan-50">
      <h1 className="mb-1 text-2xl font-bold tracking-[0.3em] text-cyan-300">JARVIS ALERTS</h1>
      <p className="mb-4 text-sm text-cyan-100/60">
        Example rule only: 9 average crosses 21 average, with an RSI filter. Alerts are not predictions.
        Keep this tab open.
      </p>
      <div className="mb-4 flex flex-wrap gap-3">
        <input value={symbol} onChange={(e) => setSymbol(e.target.value.toUpperCase())}
          className="rounded-lg border border-cyan-400/30 bg-black/40 px-3 py-2 text-sm" />
        <select value={interval} onChange={(e) => setIntervalValue(e.target.value)}
          className="rounded-lg border border-cyan-400/30 bg-black/40 px-3 py-2 text-sm">
          {["1m", "3m", "5m", "15m", "30m", "1h"].map((i) => (
            <option key={i}>{i}</option>
          ))}
        </select>
        <button onClick={toggle}
          className={`rounded-lg px-5 py-2 text-sm font-semibold ${running ? "bg-red-500 text-white" : "bg-cyan-500 text-black"}`}>
          {running ? "Stop watching" : "Start watching"}
        </button>
      </div>
      {error && <div className="mb-3 text-sm text-red-300">Error: {error}</div>}
      {last && (
        <div className="mb-4 rounded-xl border border-cyan-400/20 bg-cyan-400/5 p-4 text-sm">
          <div>{last.symbol} ({last.interval}) price: {last.price}</div>
          <div>EMA9: {last.ema9.toFixed(2)} | EMA21: {last.ema21.toFixed(2)} | RSI: {last.rsi.toFixed(1)}</div>
          <div>Current signal: {last.signal}</div>
        </div>
      )}
      <div className="text-xs uppercase tracking-[0.3em] text-cyan-300">Alert history</div>
      <div className="mt-2 space-y-2 text-sm">
        {log.length === 0 && <div className="text-cyan-100/40">No alerts yet.</div>}
        {log.map((e, i) => (
          <div key={i} className="rounded-lg border border-cyan-400/20 p-2">
            {e.time} - {e.text}
          </div>
        ))}
      </div>
    </div>
  );
}
